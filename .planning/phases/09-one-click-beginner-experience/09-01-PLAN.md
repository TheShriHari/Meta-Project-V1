# Plan 09-01: Beginner Landing Screen, One-Click Simulation, Slow-Motion Playback, and Hall–Petch Line Chart

## Goal

Add a beginner-friendly default landing experience to HPSim: a clean screen with a title, subtitle, and a single **Simulate Now** button that automatically starts a slow-motion 3D edge-dislocation animation using the existing `buildDemo` / `SimulationController` / `Viewer` engine, alongside a simple Hall–Petch line chart drawn from `runHallPetchSweep` + `CALIBRATED_MATERIALS.aluminium`. The existing research dashboard (Phases 1–8 controls) becomes the "Advanced Simulation" mode, still fully accessible via a link. No scientific engine files are modified.

---

## Architecture Constraints (enforce throughout)

- **Do not modify**: `web/src/core/*`, `web/src/render/*`, `web/src/model/demo.ts`, any `*.test.mjs`, any Python file.
- **Slow motion** = `speed = 0.2` in `framesToSteps()` call. Do **not** change `dt_s`, `M_d`, `tau_c_pa`, or any material parameter.
- **Hall–Petch data** must come from `runHallPetchSweep(inputs)` with `CALIBRATED_MATERIALS.aluminium` — do not hardcode numbers.
- **Dislocation type**: Edge only. Label it "Edge dislocation (⊥)" and note screw is not implemented.
- **No new dependencies**: No charting library, no router library. Use inline SVG for the chart.

```tsx
/**
 * web/src/ui/BeginnerScreen.tsx
 * Beginner-friendly landing screen for HPSim.
 * Uses existing buildDemo / SimulationController / Viewer APIs unchanged.
 * Scientific integrity: edge-only animation labelled correctly; slow motion via speed multiplier only.
 */
import { useEffect, useRef, useState, useCallback } from "react";
import { Viewer, LAYERS } from "../render/viewer.ts";
import { buildDemo, bestSystemIndex, LOADING_AXES } from "../model/demo.ts";
import { framesToSteps } from "../core/playback.ts";

const BEGINNER_AXIS_KEY = "[001]";                    // [001] loading: clean single-slip FCC
const BEGINNER_STRESS_MPA = 40;                       // above tau_c (5 MPa) → dislocation glides
const BEGINNER_CELLS = 5;
const SLOW_SPEED = 0.2;                               // 0.2× wall-clock → visibly slow motion
const BASE_STEPS_PER_SECOND = 60;
const MAX_BEGINNER_HISTORY = 10000;

interface BeginnerScreenProps {
  onGoAdvanced: () => void;
}

export function BeginnerScreen({ onGoAdvanced }: BeginnerScreenProps) {
  const hostRef = useRef<HTMLDivElement>(null);
  const viewerRef = useRef<Viewer | null>(null);
  const rafRef = useRef<number>(0);
  const simRef = useRef<ReturnType<typeof buildDemo> | null>(null);
  const [simState, setSimState] = useState<"idle" | "running" | "done">("idle");
  const [runCount, setRunCount] = useState(0);  // increments on each fresh run

  // Mount viewer once
  useEffect(() => {
    const v = new Viewer(hostRef.current!);
    viewerRef.current = v;
    return () => {
      v.dispose();
      viewerRef.current = null;
    };
  }, []);

  // Build and start simulation on each runCount change (triggered by button)
  useEffect(() => {
    if (runCount === 0) return;  // skip on initial mount
    const viewer = viewerRef.current;
    if (!viewer) return;

    // Cancel any in-flight animation
    cancelAnimationFrame(rafRef.current);

    // Build demo with auto-selected slip system
    const axis = LOADING_AXES[BEGINNER_AXIS_KEY];
    const systemIndex = bestSystemIndex(axis);
    const demo = buildDemo({
      cells: BEGINNER_CELLS,
      systemIndex,
      axisKey: BEGINNER_AXIS_KEY,
      appliedStress_pa: BEGINNER_STRESS_MPA * 1e6,
    });
    simRef.current = demo;

    // Load structure into viewer
    viewer.loadStructure(demo);

    // Set visible layers: atoms, slipPlane, core only
    for (const layer of LAYERS) {
      viewer.setLayer(layer, layer === "atoms" || layer === "slipPlane" || layer === "core");
    }
    viewer.setView("front");
    viewer.setCorePosition(demo.controller.state.pos_m);

    setSimState("running");

    // Slow-motion playback loop using framesToSteps — speed multiplier only, no physics change
    let acc = 0;
    let last = performance.now();
    const c = demo.controller;

    const tick = (now: number) => {
      const frameDt = Math.min((now - last) / 1000, 0.1);
      last = now;
      const r = framesToSteps(acc, frameDt, SLOW_SPEED, BASE_STEPS_PER_SECOND);
      acc = r.accumulator;
      for (let k = 0; k < r.steps; k++) {
        c.step();
        if (c.state.status === "at_domain_limit" || c.history.length >= MAX_BEGINNER_HISTORY) {
          viewer.setCorePosition(c.state.pos_m);
          setSimState("done");
          return;  // stop RAF
        }
      }
      if (r.steps > 0) {
        viewer.setCorePosition(c.state.pos_m);
      }
      rafRef.current = requestAnimationFrame(tick);
    };
    rafRef.current = requestAnimationFrame(tick);

    return () => cancelAnimationFrame(rafRef.current);
  }, [runCount]);

  const handleSimulateNow = useCallback(() => {
    setSimState("idle");
    setRunCount(n => n + 1);
  }, []);

  return (
    <div className="beginner-root">
      {/* Header */}
      <header className="beginner-header">
        <h1 className="beginner-title">HPSim — See How Dislocations Move</h1>
        <p className="beginner-subtitle">
          Watch how tiny defects move through crystals and how grain size affects material strength.
        </p>
      </header>

      {/* Main content: 3D viewer + chart side by side */}
      <main className="beginner-main">
        {/* Left: 3D Simulation area */}
        <section className="beginner-sim-panel">
          <div ref={hostRef} className="beginner-canvas" />
          <div className="beginner-sim-controls">
            <button
              className="beginner-simulate-btn"
              id="btn-simulate-now"
              onClick={handleSimulateNow}
            >
              {simState === "running" ? "▶ Running…" : "Simulate Now"}
            </button>
            {simState === "done" && (
              <span className="beginner-done-msg">
                ✓ Simulation complete — click again for a fresh run
              </span>
            )}
          </div>
          <div className="beginner-dislocation-label">
            <span className="beginner-type-badge">Edge dislocation (⊥)</span>
            <span className="beginner-engine-note">
              Screw dislocation visualization: not yet implemented (edge field only)
            </span>
          </div>
        </section>

        {/* Right: Hall–Petch chart — rendered by HallPetchChart component (Task T2) */}
        <section className="beginner-chart-panel">
          {/* Placeholder filled in T2 */}
          <div className="beginner-chart-placeholder" id="hall-petch-chart-area">
            <p style={{ color: "#6b7fa3", fontSize: "13px", textAlign: "center", paddingTop: "40px" }}>
              Hall–Petch chart loading…
            </p>
          </div>
        </section>
      </main>

      {/* Footer navigation */}
      <footer className="beginner-footer">
        <button
          className="beginner-advanced-link"
          id="btn-go-advanced"
          onClick={onGoAdvanced}
        >
          Advanced Simulation →
        </button>
        <span className="beginner-footer-note">
          Full scientific controls: Schmid glide, grain-boundary transmission, pile-up, micrograph stereology
        </span>
      </footer>
    </div>
  );
}
```

**Step 1: Create `web/src/ui/HallPetchChart.tsx`**

```tsx
/**
 * web/src/ui/HallPetchChart.tsx
 * Simple inline SVG Hall–Petch line chart for the beginner screen.
 * Data source: runHallPetchSweep + CALIBRATED_MATERIALS.aluminium (Phase 6, V-09/V-10).
 * Provenance: Calibrated model trend only — NOT raw experimental data.
 * No external charting library. Pure SVG.
 */
import { useMemo } from "react";
import { runHallPetchSweep, CALIBRATED_MATERIALS } from "../core/hallpetch.ts";

const AL = CALIBRATED_MATERIALS.aluminium;

const GRAIN_SIZES_M: number[] = (() => {
  const pts = 16;
  const logMin = Math.log10(10e-6);   // 10 μm
  const logMax = Math.log10(400e-6);  // 400 μm
  return Array.from({ length: pts }, (_, i) =>
    Math.pow(10, logMin + (i / (pts - 1)) * (logMax - logMin))
  );
})();

export function HallPetchChart() {
  const sweep = useMemo(() =>
    runHallPetchSweep({
      muPa: AL.muPa,
      nu: AL.nu,
      burgersM: AL.burgersM,
      tauCPa: AL.tauCPa,
      taylorFactor: AL.taylorFactor,
      tauStarRefPa: AL.calibratedTauStarRefPa,
      grainSizesM: GRAIN_SIZES_M,
      sigma0OffsetPa: AL.calibrationFit.sigma0Pa,
      bootstrapSamples: 50,   // fewer samples for fast render
      seed: 42,
    }),
  []);

  // Convert to display units:
  // X = invSqrtDM05 (m^-0.5) → μm^-0.5 = × 1e-3
  // Y = sigmaYPa (Pa)         → MPa = ÷ 1e6
  const points = sweep.points.map(p => ({
    x: p.invSqrtDM05 * 1e-3,    // μm^{-0.5}
    y: p.sigmaYPa / 1e6,         // MPa
  }));

  // SVG layout
  const W = 340, H = 240;
  const PAD = { top: 28, right: 20, bottom: 44, left: 56 };
  const plotW = W - PAD.left - PAD.right;
  const plotH = H - PAD.top - PAD.bottom;

  const xMin = 0;
  const xMax = Math.max(...points.map(p => p.x)) * 1.05;
  const yMin = Math.min(...points.map(p => p.y)) * 0.92;
  const yMax = Math.max(...points.map(p => p.y)) * 1.05;

  const toSvgX = (x: number) => PAD.left + ((x - xMin) / (xMax - xMin)) * plotW;
  const toSvgY = (y: number) => PAD.top + (1 - (y - yMin) / (yMax - yMin)) * plotH;

  const polylinePoints = points.map(p => `${toSvgX(p.x).toFixed(1)},${toSvgY(p.y).toFixed(1)}`).join(" ");

  // Y-axis ticks (4 ticks)
  const yTicks = [0, 1, 2, 3].map(i => yMin + (i / 3) * (yMax - yMin));
  // X-axis ticks (4 ticks)
  const xTicks = [0, 1, 2, 3].map(i => xMin + (i / 3) * (xMax - xMin));

  return (
    <div className="hp-chart-wrapper">
      <h3 className="hp-chart-title">Hall–Petch Relationship</h3>
      <p className="hp-chart-subtitle">
        σ<sub>y</sub> = σ<sub>0</sub> + k<sub>y</sub> · d<sup>−½</sup>
      </p>
      <svg
        viewBox={`0 0 ${W} ${H}`}
        width="100%"
        style={{ maxWidth: W, display: "block" }}
        aria-label="Hall-Petch line chart: yield strength vs inverse square root of grain size"
        role="img"
      >
        {/* Background */}
        <rect x={PAD.left} y={PAD.top} width={plotW} height={plotH}
          fill="rgba(15,25,50,0.6)" rx="4" />

        {/* Grid lines */}
        {yTicks.map((y, i) => (
          <line key={i}
            x1={PAD.left} y1={toSvgY(y)}
            x2={PAD.left + plotW} y2={toSvgY(y)}
            stroke="#1f2a4d" strokeWidth="1" />
        ))}
        {xTicks.map((x, i) => (
          <line key={i}
            x1={toSvgX(x)} y1={PAD.top}
            x2={toSvgX(x)} y2={PAD.top + plotH}
            stroke="#1f2a4d" strokeWidth="1" />
        ))}

        {/* Hall-Petch line */}
        <polyline
          points={polylinePoints}
          fill="none"
          stroke="#7c5cff"
          strokeWidth="2.5"
          strokeLinejoin="round"
        />

        {/* Axes */}
        <line x1={PAD.left} y1={PAD.top} x2={PAD.left} y2={PAD.top + plotH}
          stroke="#60a5fa" strokeWidth="1.5" />
        <line x1={PAD.left} y1={PAD.top + plotH} x2={PAD.left + plotW} y2={PAD.top + plotH}
          stroke="#60a5fa" strokeWidth="1.5" />

        {/* Y-axis label */}
        <text
          x={14} y={PAD.top + plotH / 2}
          textAnchor="middle"
          fill="#9db4c8"
          fontSize="10"
          transform={`rotate(-90, 14, ${PAD.top + plotH / 2})`}
        >
          σ_y (MPa)
        </text>

        {/* X-axis label */}
        <text
          x={PAD.left + plotW / 2}
          y={H - 6}
          textAnchor="middle"
          fill="#9db4c8"
          fontSize="10"
        >
          d⁻¹/² (μm⁻¹/²)
        </text>

        {/* Y tick labels */}
        {yTicks.map((y, i) => (
          <text key={i}
            x={PAD.left - 5} y={toSvgY(y) + 4}
            textAnchor="end" fill="#64748b" fontSize="9">
            {y.toFixed(0)}
          </text>
        ))}

        {/* X tick labels */}
        {xTicks.map((x, i) => (
          <text key={i}
            x={toSvgX(x)} y={PAD.top + plotH + 13}
            textAnchor="middle" fill="#64748b" fontSize="9">
            {x.toFixed(3)}
          </text>
        ))}
      </svg>

      {/* Provenance note — scientifically mandatory */}
      <p className="hp-chart-provenance" id="hp-chart-provenance">
        Calibrated model trend — Al, Hansen &amp; Ralph (1982).
        Not raw experimental data. τ*<sub>ref</sub> calibrated per V-15.
      </p>
      <p className="hp-chart-material">
        Material: Aluminium (FCC) · σ<sub>0</sub> = {(AL.calibrationFit.sigma0Pa / 1e6).toFixed(0)} MPa ·
        k<sub>y</sub> = {(AL.calibrationFit.kyPaM05 / 1000).toFixed(0)} MPa·μm<sup>½</sup>
      </p>
    </div>
  );
}
```
